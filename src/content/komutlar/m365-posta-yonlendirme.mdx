ozet: "Kurum dışına yönlendirme, ele geçirilmiş hesabın ilk işaretidir."
konu: "m365"
etiketler: ["m365", "guvenlik", "exchange"]
kod: |
  Get-Mailbox -ResultSize Unlimited |
    Where-Object { $_.ForwardingSmtpAddress -or $_.ForwardingAddress } |
    Select-Object DisplayName, PrimarySmtpAddress, ForwardingSmtpAddress,
                  DeliverToMailboxAndForward
dikkat: "Gelen kutusu kuralıyla yapılan yönlendirme burada görünmez; `Get-InboxRule` ile ayrıca tarayın."
